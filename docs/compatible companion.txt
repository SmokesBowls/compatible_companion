# zw_sidecar_fastapi_capsule.py
# FastAPI ZW sidecar: validate/emit + compressed memory + signed "ZW Capsule" import/export + consent ledger
from fastapi import FastAPI, Body, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import StreamingResponse
from pydantic import BaseModel, Field, validator
from typing import List, Optional, Dict, Any
from datetime import datetime, timezone
import os, json, re, hashlib, base64

# --- Ed25519 signing (PyNaCl) ----------------------------------------------
try:
    from nacl.signing import SigningKey, VerifyKey
    from nacl.exceptions import BadSignatureError
except Exception as e:
    raise SystemExit("Missing dependency: pip install pynacl") from e

APP_PORT = int(os.getenv("ZW_PORT", "7077"))
DB_PATH   = os.getenv("ZW_DB",   "./zw_mem.jsonl")
KEYS_PATH = os.getenv("ZW_KEYS", "./zw_keys.json")

TTL_DAYS = {"open_loop": 14, "style": 180}  # canon/character: no TTL
VALID_SCOPES = {"canon","character","open_loop","style"}

def now_iso() -> str:
    return datetime.now(timezone.utc).isoformat()

def iso_to_ts(s: str) -> float:
    try:
        if s.endswith("Z"): s = s[:-1] + "+00:00"
        return datetime.fromisoformat(s).timestamp()
    except Exception:
        return datetime.now(timezone.utc).timestamp()

def normalize(s: str) -> str:
    return re.sub(r"\W+", "", (s or "").lower()).strip()

def compute_hash(summary: str, bullets: List[str]) -> str:
    base = normalize(summary) + "|" + "|".join(normalize(b) for b in (bullets or []))
    return hashlib.sha1(base.encode()).hexdigest()

# Keys: create/load owner signing key
def load_keys():
    if os.path.exists(KEYS_PATH):
        data = json.load(open(KEYS_PATH,"r",encoding="utf-8"))
        sk = SigningKey(base64.b64decode(data["sk_b64"]))
        vk = VerifyKey(base64.b64decode(data["pk_b64"]))
        return sk, vk, data["did"]
    # generate new
    sk = SigningKey.generate()
    vk = sk.verify_key
    pk_b64 = base64.b64encode(bytes(vk)).decode()
    sk_b64 = base64.b64encode(bytes(sk)).decode()
    did = f"did:zw:{pk_b64}"  # simple, portable DID (base64 pubkey)
    json.dump({"pk_b64":pk_b64,"sk_b64":sk_b64,"did":did}, open(KEYS_PATH,"w",encoding="utf-8"), indent=2)
    return sk, vk, did

SIGNING_KEY, VERIFY_KEY, OWNER_DID = load_keys()

def sign_bytes(b: bytes) -> str:
    sig = SIGNING_KEY.sign(b).signature
    return base64.b64encode(sig).decode()

def verify_bytes(b: bytes, sig_b64: str, pk_b64: str) -> bool:
    try:
        VerifyKey(base64.b64decode(pk_b64)).verify(b, base64.b64decode(sig_b64))
        return True
    except BadSignatureError:
        return False

# --- Minimal persistence ----------------------------------------------------
def persist(obj: Dict[str, Any]) -> None:
    with open(DB_PATH,"a",encoding="utf-8") as f:
        f.write(json.dumps(obj, ensure_ascii=False) + "\n")

def load_db() -> List[Dict[str, Any]]:
    if not os.path.exists(DB_PATH): return []
    out = []
    for line in open(DB_PATH,"r",encoding="utf-8"):
        line=line.strip()
        if not line: continue
        try: out.append(json.loads(line))
        except: pass
    return out

RECORDS = load_db()
VIEW: Dict[str, Dict[str, Any]] = {}  # latest mem unit per id
for rec in RECORDS:
    if isinstance(rec, dict) and "id" in rec and rec.get("scope") in VALID_SCOPES:
        VIEW[rec["id"]] = rec

def expire_if_needed():
    from time import time
    expired = []
    now = time()
    for uid, unit in list(VIEW.items()):
        ttl = TTL_DAYS.get(unit.get("scope"))
        if not ttl: continue
        ts = iso_to_ts(unit.get("ts") or unit.get("timestamp") or unit.get("updated_at") or unit.get("created_at") or now_iso())
        if (now - ts) > ttl*24*3600:
            expired.append(uid)
    for uid in expired:
        scope = VIEW[uid]["scope"]
        VIEW.pop(uid, None)
        persist({"op":"expire","id":uid,"scope":scope,"ts":now_iso()})
    return expired

    @validator("summary")   def _limsum(cls,v): return v[:240] if v else v
    @validator("bullets", pre=True)   def _limbul(cls,v): return list(v or [])[:5]
    @validator("entities", pre=True)  def _liment(cls,v): return list(v or [])[:10]

@app.post("/mem/store")
def mem_store(unit: MemUnit):
    u = unit.dict()
    if not u.get("ts"):   u["ts"]   = now_iso()
    if not u.get("hash"): u["hash"] = compute_hash(u.get("summary",""), u.get("bullets") or [])
    existing = VIEW.get(u["id"])
    if existing and existing.get("hash") == u["hash"]:
        return {"ok":True, "dedup":True}
    if existing:
        persist({"op":"replace","prev":existing,"next":u,"ts":now_iso()})
    VIEW[u["id"]] = u
    persist(u)
    expired = expire_if_needed()
    return {"ok":True, "replacedId": (existing and u["id"]) or None, "expired": expired}

@app.post("/mem/find")
def mem_find(req: MemFindReq):
    Q = (req.q or "").lower()
    results = []
    for m in VIEW.values():
        if req.scope and m.get("scope") != req.scope: continue
        if req.tags:
            tset = set(m.get("tags") or [])
            if not any(t in tset for t in req.tags): continue
        if req.entities:
            eset = set(m.get("entities") or [])
            if not any(e in eset for e in req.entities): continue
        if Q:
            hay = " ".join([m.get("id",""), m.get("topic","") or "", m.get("summary","") or "", * (m.get("bullets") or [])]).lower()
            if Q not in hay: continue
        results.append(m)
    scope_rank = {"canon":0,"character":1,"style":2,"open_loop":3}
    results.sort(key=lambda m: (-iso_to_ts(m.get("ts","")), scope_rank.get(m.get("scope"),9)))
    return {"ok":True,"count":len(results),"results":results[:50]}

# --- Capsule export/import --------------------------------------------------
class CapsuleExportReq(BaseModel):
    profile: Dict[str, Any] = {}                 # persona/style/goals to embed
    skills: List[Skill] = []                     # exported skills (names, urls, strict)
    policy: Policy = Policy()

@app.post("/capsule/export")
def capsule_export(req: CapsuleExportReq):
    units = list(VIEW.keys())
    counts = {"canon":0,"character":0,"style":0,"open_loop":0}
    for u in VIEW.values():
        counts[u["scope"]] = counts.get(u["scope"],0)+1
    capsule = Capsule(
        identity={"did":OWNER_DID, "pk_b64": base64.b64encode(bytes(VERIFY_KEY)).decode()},
        profile=req.profile,
        memory_index={"units": units, "counts": counts},
        skills=req.skills,
        policy=req.policy
    ).dict()
    # hash + sign
    capsule_bytes = json.dumps(capsule, ensure_ascii=False, separators=(",",":")).encode()
    sha = hashlib.sha256(capsule_bytes).hexdigest()
    sig = sign_bytes(capsule_bytes)
    capsule["signatures"] = {"capsule_sha256": sha, "signer_did": OWNER_DID, "sig_b64": sig}
    return capsule

class CapsuleImportReq(BaseModel):
    capsule: Dict[str, Any]
    transfer_signature_b64: Optional[str] = None  # optional second signature by previous owner

@app.post("/capsule/import")
def capsule_import(req: CapsuleImportReq):
    cap = req.capsule
    sig = cap.get("signatures",{}).get("sig_b64")
    pk_b64 = cap.get("identity",{}).get("pk_b64")
    if not (sig and pk_b64):
        raise HTTPException(400, "capsule missing signature/pk_b64")
    # verify payload integrity
    tmp = dict(cap); tmp.pop("signatures",None)
    b = json.dumps(tmp, ensure_ascii=False, separators=(",",":")).encode()
    if not verify_bytes(b, sig, pk_b64):
        raise HTTPException(400, "invalid capsule signature")
    # adopt profile/skills/policy (store as ops for transparency)
    persist({"op":"capsule_import_meta","ts":now_iso(),"identity":cap.get("identity"),"profile":cap.get("profile"),"skills":cap.get("skills"),"policy":cap.get("policy")})
    # NOTE: memory units themselves are NOT embedded in capsule by default.
    # If you include units separately, call /mem/import later with those units.

    return {"ok":True, "adopted": ["identity","profile","skills","policy"]}

# Optional bulk import of units (same as previous design)
class MemImportReq(BaseModel):
    units: List[MemUnit]

@app.post("/mem/import")
def mem_import(req: MemImportReq):
    inserted = replaced = deduped = 0
    for unit in req.units:
        u = unit.dict()
        if not u.get("ts"):   u["ts"]   = now_iso()
        if not u.get("hash"): u["hash"] = compute_hash(u.get("summary",""), u.get("bullets") or [])
        existing = VIEW.get(u["id"])
        if existing and existing.get("hash")==u["hash"]:
            deduped += 1; continue
        if existing:
            persist({"op":"replace","prev":existing,"next":u,"ts":now_iso()})
            replaced += 1
        else:
            inserted += 1
        VIEW[u["id"]] = u
        persist(u)
    expired = expire_if_needed()
    return {"ok":True,"inserted":inserted,"replaced":replaced,"deduped":deduped,"expired":expired}
