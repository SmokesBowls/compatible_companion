2. P1 — Snapshot Checkpointing
Replay currently walks the full JSONL log from genesis. At scale that is O(N) startup time. Snapshots make it O(recent) — replay only from the last checkpoint forward. The full log remains for audit. Nothing is deleted.

2.1 New File: cc/compact.py
Create this file. It has two public functions only.

# cc/compact.py
"""
Snapshot checkpointing for Compatible Companion.

compact_log(log_path, capsule_path, key_manager=None)
    Reads the current receipt log. Exports a capsule snapshot to capsule_path.
    The capsule becomes the official checkpoint. The log is NOT modified.
    If key_manager provided, capsule is signed.

replay_from_snapshot(rt, capsule_path, log_path, key_manager=None)
    Imports the snapshot into rt (AgentRuntime).
    Then replays only log entries AFTER the snapshot's log_head.
    Result: same state as full replay from genesis, but O(recent) cost.
"""
import json
from cc.capsule import export_capsule, import_capsule

def compact_log(db_path: str, capsule_path: str, key_manager=None) -> str:
    """
    Write a capsule snapshot from db_path to capsule_path.
    Returns the log_head hash embedded in the capsule.
    """
    export_capsule(db_path, capsule_path, key_manager=key_manager)
    with open(capsule_path) as f:
        caps = json.load(f)
    return caps.get('log_head', '')   # the receipt hash at snapshot time

def replay_from_snapshot(rt, capsule_path: str, log_path: str,
                          key_manager=None) -> None:
    """
    1. Import snapshot into rt (restores committed state).
    2. Read log_path. Skip all entries with receipt_id <= snapshot log_head.
    3. Replay only the tail entries via rt.apply_receipt(receipt).
    """
    import_capsule(capsule_path, rt.db_path, key_manager=key_manager)

    with open(capsule_path) as f:
        caps = json.load(f)
    checkpoint_head = caps.get('log_head', '')

    with open(log_path) as f:
        entries = [json.loads(l) for l in f if l.strip()]

    # Find the first entry AFTER the checkpoint
    past_checkpoint = False
    for entry in entries:
        if entry.get('receipt_id') == checkpoint_head:
            past_checkpoint = True
            continue
        if past_checkpoint:
            rt.apply_receipt(entry)   # replay tail only

IMPORTANT: export_capsule must embed the current log_head (the most recent committed receipt_id) in the capsule JSON so replay_from_snapshot knows where to cut the log. If log_head is not already in the capsule schema, add it now. This is the only schema change in Week 8.

2.2 apply_receipt on AgentRuntime
replay_from_snapshot calls rt.apply_receipt(receipt). This method must exist. It is the atomic unit of replay — given a committed receipt dict, re-apply its writes to the DB without going through PLAN→ACT→VERIFY. This is replay-only, not a new write path.

# In cc/runtime.py — add this method to AgentRuntime
def apply_receipt(self, receipt: dict) -> None:
    """
    Replay a committed receipt directly into storage.
    Called only during log replay. Never called from production write paths.
    Raises ValueError if receipt outcome is not COMMIT.
    """
    if receipt.get('outcome') != 'COMMIT':
        return   # skip FAIL receipts — they left no state

    # Re-apply staged writes from the receipt
    for unit in receipt.get('staged_units', []):
        self.memory.mem_store(
            unit_id=unit['unit_id'],
            scope=unit['scope'],
            body=unit['body'],
            tags=unit.get('tags', []),
            ttl_expires_at=unit.get('ttl_expires_at'),
        )

    # Update state hash
    self.last_state_hash = receipt.get('state_hash', self.last_state_hash)

NOTE: apply_receipt requires that COMMIT receipts contain staged_units in the JSON. Check your receipt schema now. If staged_units is not serialised into COMMIT receipts, add it. This is required for both snapshot replay and full replay correctness.

                plan={'action':'mem_store', f'unit_id': f'u{i}',

                act_fn=lambda p: {**p}, invariants=[],
            )
        full_hash = rt.last_state_hash

        # Take snapshot after 3 units
        compact_log(db, snap)

        # Write one more unit AFTER snapshot
        rt.run_cycle(
            plan={'action':'mem_store','unit_id':'u3','scope':'core',
                  'tags':[],'body':{'summary':'post-snapshot'}},
            act_fn=lambda p: {**p}, invariants=[],
        )
        final_hash = rt.last_state_hash

        # Full replay from genesis
        rt_full = AgentRuntime(db_path=os.path.join(tmp,'full.db'),
                               log_path=os.path.join(tmp,'full.jsonl'))

        ids = [u.get('unit_id') for u in all_units]
        assert all(f'u{i}' in ids for i in range(4)), f'all units must be present: {ids}'

3.1 New File: cc/server.py
# cc/server.py
"""
TCP transport layer for Compatible Companion MCP server.

start_tcp_server(rt, key_manager, host='127.0.0.1', port=7700)
    Starts an asyncio TCP server wrapping the MCP handlers.
    Each connection gets a framed JSON request/response protocol.
    Server runs until cancelled.

Protocol (newline-delimited JSON):
  Request:  {"tool": "mem_find", "args": {"scope": "core"}}
  Response: {"ok": true, "result": [...]}
  Error:    {"ok": false, "error": "..."}
"""
import asyncio, json
from cc.mcp_adapter import build_server

async def _handle_client(rt, handlers, reader, writer):
    addr = writer.get_extra_info('peername')
    try:
        while True:
            line = await reader.readline()
            if not line:
                break
            try:
                req = json.loads(line.decode())
                tool = req.get('tool')
                args = req.get('args', {})
                if tool not in handlers:
                    raise ValueError(f'unknown tool: {tool}')
                result = await handlers[tool](tool, args)
                payload = json.dumps({'ok': True,
                    'result': json.loads(result[0].text)}) + '\n'
            except Exception as e:
                payload = json.dumps({'ok': False, 'error': str(e)}) + '\n'
            writer.write(payload.encode())
            await writer.drain()
    finally:
        writer.close()
        await writer.wait_closed()

async def start_tcp_server(rt, key_manager,
                            host='127.0.0.1', port=7700):
    _, handlers = build_server(rt, key_manager)
    server = await asyncio.start_server(
        lambda r, w: _handle_client(rt, handlers, r, w),
        host, port
    )
    async with server:
        await server.serve_forever()

def test_30_tcp_transport_mem_find():

    Two-process proof: server in background task, client connects via TCP.
    mem_find via TCP returns correct results. No shared memory.
    """
    import tempfile, os, asyncio, json
    from cc.runtime import AgentRuntime
    from cc.identity import generate_keypair, KeyManager
    from cc.server import start_tcp_server
    import shutil

    async def run():
        with tempfile.TemporaryDirectory() as tmp:
            sk, vk = generate_keypair()
            kp = os.path.join(tmp, 'id.key')
            shutil.copy2('cc_identity.key', kp)
            km = KeyManager(kp)

            db  = os.path.join(tmp, 'server.db')
            log = os.path.join(tmp, 'server.jsonl')
            rt  = AgentRuntime(db_path=db, log_path=log)
            rt.run_cycle(
                plan={'action':'mem_store','unit_id':'tcp_unit',
                      'scope':'core','tags':[],'body':{'summary':'tcp test'}},
                act_fn=lambda p: {**p}, invariants=[],
            )

            # Start server in background
            server_task = asyncio.create_task(
                start_tcp_server(rt, km, host='127.0.0.1', port=7700)
            )
            await asyncio.sleep(0.1)   # let server bind

            # Client connects via TCP
            reader, writer = await asyncio.open_connection('127.0.0.1', 7700)
            req = json.dumps({'tool':'mem_find','args':{'scope':'core'}}) + '\n'
            writer.write(req.encode())
            await writer.drain()

            line = await reader.readline()
            resp = json.loads(line.decode())

            writer.close()
            await writer.wait_closed()
            server_task.cancel()
            try: await server_task
            except asyncio.CancelledError: pass

            assert resp['ok'] is True, f'TCP mem_find must succeed: {resp}'
            ids = [u.get('unit_id') for u in resp['result']]
            assert 'tcp_unit' in ids, f'tcp_unit must be in results: {ids}'

    asyncio.run(run())

7. Week 9 Preview (Do Not Build Yet)
Week 9 is where Compatible Companion becomes something you can actually deploy. Two items:

TLS + auth on TCP
The TCP server accepts a shared secret or certificate. Unauthenticated connections are rejected. This is the minimum bar for anything that touches a real network.
Multi-client session isolation
Two clients connect simultaneously. Their reads and writes are isolated — one client's staged writes do not bleed into the other's VERIFY phase. This requires a per-connection transaction context.

Week 9 is listed here for orientation only. Do not implement it in Week 8. Scope is a wall.
