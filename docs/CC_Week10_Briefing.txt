2. P1 — Async/Sync Decoupling in MCP Adapter
The call_tool method is async. Everything it calls is blocking SQLite. The event loop freezes on every DB operation. The 5-second timeout in memory.py is the admission — the system was already experiencing contention. Fix it at the source.

2.1 mcp_adapter.py — run_in_executor pattern
# In cc/mcp_adapter.py — update the handler methods
# Wrap ALL blocking calls in run_in_executor
import asyncio

async def _handle_mem_find(rt, tool_name, args):
    scope = args.get('scope', 'core')
    loop = asyncio.get_event_loop()
    # Offload blocking SQLite call to thread pool
    units = await loop.run_in_executor(
        None,  # default ThreadPoolExecutor
        lambda: rt.memory.mem_find(scope=scope)
    )
    import json
    from mcp.types import TextContent
    return [TextContent(type='text', text=json.dumps(units))]

async def _handle_mem_store(rt, km, tool_name, args):
    loop = asyncio.get_event_loop()
    # run_cycle is the blocking call — offload it
    result = await loop.run_in_executor(
        None,
        lambda: rt.run_cycle(
            plan={
                'action':    'mem_store',
                'unit_id':   args.get('unit_id'),
                'scope':     args.get('scope', 'core'),
                'tags':      args.get('tags', []),
                'body':      args.get('body', {}),
                'ttl_expires_at': args.get('ttl_expires_at'),
            },
            act_fn=lambda p: {**p},
            invariants=[],
        )
    )
    from mcp.types import TextContent
    import json
    return [TextContent(type='text', text=json.dumps({'outcome': result}))]

IMPORTANT: run_in_executor moves the blocking call to a background thread.
The event loop stays responsive — it can service heartbeats, cancellation requests,
and other connections while the DB operation completes.

This does NOT give you concurrent writes. SQLite WAL still serializes writers.
What you gain: the server stops freezing. Requests queue instead of timing out.
That is the difference between a robust service and a fragile one.

3. P2 — Snapshot Freeze + Explicit Paths

Current compact_log derives the db_path by replacing .jsonl with .db in the log path.
This breaks if logs and DB are on different drives, differently named, or in different dirs.

Worse: there is a race window between reading the log head and exporting the capsule.
A write that lands in that window creates a zombie state — data in the snapshot
that is not accounted for in the log head recorded. Replay integrity breaks silently.
This will not appear in tests. It will appear six months later during an incident audit.

3.1 compact.py — explicit paths + freeze
# cc/compact.py — replace compact_log with this

import json, threading
from cc.capsule import export_capsule, import_capsule

def compact_log(db_path: str, log_path: str,
                capsule_path: str,
                rt=None,
                key_manager=None) -> str:
    """
    Write a capsule snapshot.
    db_path and log_path are REQUIRED — no path inference.
    If rt is provided, acquires rt._compact_lock to freeze run_cycle
    during export. This closes the race window.
    Returns the log_head hash embedded in the capsule.
    """
    # Acquire freeze lock if runtime provided
    lock = getattr(rt, '_compact_lock', None) if rt else None

    if lock is not None:
        lock.acquire()
    try:
        export_capsule(db_path, capsule_path, key_manager=key_manager)
    finally:
        if lock is not None:
            lock.release()

    with open(capsule_path) as f:
        caps = json.load(f)
    return caps.get('log_head', '')

3.2 runtime.py — _compact_lock
# In cc/runtime.py — AgentRuntime.__init__, add:

def test_37_snapshot_freeze_no_zombie():
    """
    compact_log with explicit paths and freeze lock.
    A concurrent run_cycle cannot commit between log-head read and export.
    """
    import tempfile, os, threading, time
    from cc.runtime import AgentRuntime
    from cc.compact import compact_log, replay_from_snapshot
    from cc.policy_store import seal_policy

    with tempfile.TemporaryDirectory() as tmp:
        db   = os.path.join(tmp, 'main.db')
        log  = os.path.join(tmp, 'receipts.jsonl')
        snap = os.path.join(tmp, 'snap.json')
        rt   = AgentRuntime(db_path=db, log_path=log)
        seal_policy(rt.policy_store, log)
        rt._recheck_policy_integrity()

        # Write two units
        for i in range(2):
            rt.run_cycle(
                plan={'action':'mem_store', 'unit_id':f'u{i}',
                      'scope':'core','tags':[],'body':{'summary':f'u{i}'}},
                act_fn=lambda p: {**p}, invariants=[])

        # Take snapshot with explicit paths and freeze
        log_head = compact_log(db, log, snap, rt=rt)
        assert log_head, 'compact_log must return a log_head'

        # Write one more after snapshot
        rt.run_cycle(
            plan={'action':'mem_store','unit_id':'u2','scope':'core',
                  'tags':[],'body':{'summary':'post-snap'}},
            act_fn=lambda p: {**p}, invariants=[])

        final_hash = rt.last_state_hash

        # Replay from snapshot must reach final_hash
        rt2 = AgentRuntime(db_path=os.path.join(tmp,'r.db'),
                           log_path=os.path.join(tmp,'r.jsonl'))
        replay_from_snapshot(rt2, snap, log)
        assert rt2.last_state_hash == final_hash, \
            f'snapshot replay must match: {rt2.last_state_hash} != {final_hash}'

        # All 3 units present
        units = rt2.memory.mem_find(scope='core')
        ids = [u.get('unit_id') for u in units]
        assert all(f'u{i}' in ids for i in range(3)), f'all units must be present: {ids}'


4. P3 — Semantic MCP API — Remove run_cycle
run_cycle is the engine. It must not be the steering wheel. Exposing it as an MCP tool forces the calling agent to construct internal plan dicts, understand the CC schema, and break silently when that schema changes. Remove it. The agent calls mem_store. The adapter builds the plan.

# In cc/mcp_adapter.py
async def _handle_batch_store(rt, km, tool_name, args):
    units = args.get('units', [])
    if not units:
        from mcp.types import TextContent
        import json
        return [TextContent(type='text',
                text=json.dumps({'outcome': 'FAIL', 'error': 'no units provided'}))]

    loop = asyncio.get_event_loop()
    results = []
    for unit in units:
        result = await loop.run_in_executor(
            None,
            lambda u=unit: rt.run_cycle(
                plan={
                    'action':   'mem_store',
                    'unit_id':  u.get('unit_id'),
                    'scope':    u.get('scope', 'core'),
                    'tags':     u.get('tags', []),
                    'body':     u.get('body', {}),
                    'ttl_expires_at': u.get('ttl_expires_at'),
                },
                act_fn=lambda p: {**p},
                invariants=[],
            )
        )
        results.append({'unit_id': unit.get('unit_id'), 'outcome': result})

    from mcp.types import TextContent
    import json
    return [TextContent(type='text', text=json.dumps({'results': results}))]

5. P4 — policy_status MCP Tool
When the runtime enters locked_down mode the agent currently gets a generic error and no way to diagnose it. It cannot tell if it wrote a bad unit or if the database was tampered with. The policy_status tool gives the agent eyes. Failure becomes maintenance, not mystery.

5.1 policy_status handler
# In cc/mcp_adapter.py — add policy_status tool

# Tool definition:
{
    'name': 'policy_status',
    'description': 'Returns the current policy integrity status of the runtime.',
    'inputSchema': {'type': 'object', 'properties': {}}
}

# Handler:
async def _handle_policy_status(rt, tool_name, args):
    from mcp.types import TextContent
    import json
    status = {
        'policy_mode':    rt.policy_mode,
        'integrity_ok':   rt.policy_mode == 'normal',
        'warning':        rt.policy_integrity_warning,
        'sealed_hash':    rt.policy_store.compute_hash()
            if rt.policy_mode == 'normal' else None,
    }
    return [TextContent(type='text', text=json.dumps(status))]

policy_status is READ-ONLY. It exposes health information only.
It does NOT expose a reseal command via MCP — resealing requires operator access.
A rogue agent cannot use policy_status to bypass or restore untrusted rules.
It can only see the wound. A human operator treats it.

5.2 Acceptance Test (Test 39)
def test_39_policy_status_tool():
    """
    policy_status returns normal when sealed.
    Returns locked_down with warning when policy table is tampered.
    """
    import asyncio, tempfile, os, json, sqlite3
    from cc.runtime import AgentRuntime
    from cc.identity import generate_keypair, KeyManager
    from cc.mcp_adapter import build_server
    from cc.policy_store import seal_policy
    import shutil

    async def run():
        with tempfile.TemporaryDirectory() as tmp:
            generate_keypair()
            km = KeyManager(shutil.copy2('cc_identity.key',
                            os.path.join(tmp,'id.key')))
            db  = os.path.join(tmp,'s.db')
            log = os.path.join(tmp,'r.jsonl')
            rt  = AgentRuntime(db_path=db, log_path=log)
            rt.policy_store.add_rule('r1','test rule',"scope != 'forbidden'")
            seal_policy(rt.policy_store, log)
            rt._recheck_policy_integrity()
            _, handlers = build_server(rt, km)

            assert 'policy_status' in handlers, 'policy_status must be in handlers'

            # Normal state
            result = await handlers['policy_status']('policy_status', {})
            status = json.loads(result[0].text)
            assert status['policy_mode'] == 'normal'
            assert status['integrity_ok'] is True
            assert status['warning'] is None

            # Tamper the policy table
            conn = sqlite3.connect(db)
            conn.execute("DELETE FROM policy_rules WHERE rule_id='r1'")
            conn.commit(); conn.close()
            rt._recheck_policy_integrity()

            # Locked down state
            result2 = await handlers['policy_status']('policy_status', {})
            status2 = json.loads(result2[0].text)
            assert status2['policy_mode'] == 'locked_down'
            assert status2['integrity_ok'] is False
            assert status2['warning'] is not None
            assert 'mismatch' in status2['warning'].lower() or \
                   'hash' in status2['warning'].lower()

    asyncio.run(run())
