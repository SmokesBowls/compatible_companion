def test_26_ttl_tombstone_not_hard_deleted():
    """Expired unit is invisible to mem_find but present in raw DB."""
    import tempfile, os, sqlite3
    from datetime import datetime, timezone, timedelta
    from cc.runtime import AgentRuntime

    with tempfile.TemporaryDirectory() as tmp:
        db = os.path.join(tmp, 'test26.db')
        rt = AgentRuntime(db_path=db, log_path=os.path.join(tmp, 'r.jsonl'))
        expired = (datetime.now(timezone.utc) - timedelta(seconds=1)).isoformat()
        rt.run_cycle(
            plan={'action':'mem_store','unit_id':'u_tomb','scope':'style',
                  'tags':[],'body':{'summary':'ephemeral'},'ttl_expires_at':expired},
            act_fn=lambda p: {**p}, invariants=[],
        )
        # mem_find must not return it
        units = rt.memory.mem_find(scope='style')
        assert not any(u.get('unit_id')=='u_tomb' for u in units), 'must be invisible'

        # Raw DB query must find it (tombstone intact)
        conn = sqlite3.connect(db)
        cols = [r[1] for r in conn.execute('PRAGMA table_info(units)').fetchall()]
        id_col = 'unit_id' if 'unit_id' in cols else 'id'
        rows = conn.execute(f'SELECT {id_col} FROM units WHERE {id_col}=?',
                            ('u_tomb',)).fetchall()
        conn.close()
        assert rows, 'expired unit must be tombstoned, not hard-deleted'

        # Provenance must have a record
        conn = sqlite3.connect(db)
        prov = conn.execute(
            'SELECT op FROM provenance WHERE unit_id=?', ('u_tomb',)
        ).fetchall()
        conn.close()
        assert prov, 'provenance must record the write even if unit is expired'


3. P2 — Signed Capsule
export_capsule currently writes signature: null. Week 7 wires in signing. import_capsule verifies. Unsigned capsules are rejected. This closes the deferred stress harness test.

3.1 export_capsule Signature
export_capsule(db_path, capsule_path) must gain an optional key_manager parameter. If provided, sign the canonical capsule body and write the signature.

def export_capsule(db_path: str, capsule_path: str,
                   key_manager=None) -> None:
    """
    Export a portable capsule JSON to capsule_path.
    If key_manager is provided, signs the canonical body (JSON bytes,
    sorted keys) and writes the signature field.
    If None, writes signature: null — capsule is portable but unsigned.
    """
    # ... build capsule dict as before ...

    if key_manager is not None:
        canonical = json.dumps(capsule, sort_keys=True).encode('utf-8')
        capsule['signature'] = key_manager.sign(canonical)
    else:
        capsule['signature'] = None

    with open(capsule_path, 'w') as f:
        json.dump(capsule, f, indent=2)

3.2 import_capsule Signature Verification
def import_capsule(capsule_path: str, db_path: str,
                   key_manager=None, conflict_strategy='keep_local') -> None:
    """
    Import capsule from capsule_path into db_path.
    If key_manager provided and capsule has a signature, verify before any
    DB write. Raise on bad or missing signature.
    If key_manager is None, skip verification (unsigned import allowed).
    """
    with open(capsule_path) as f:
        capsule = json.load(f)

    if key_manager is not None:
        sig = capsule.get('signature')
        if not sig:
            raise ValueError('Capsule is unsigned. Provide unsigned key_manager=None to bypass.')
        # Verify against the canonical body (signature field excluded)
        body = {k: v for k, v in capsule.items() if k != 'signature'}
        canonical = json.dumps(body, sort_keys=True).encode('utf-8')
        from cc.identity import verify_payload
        # verify_payload raises nacl.exceptions.BadSignatureError on failure
        vk_b64 = capsule.get('public_key')
        if not vk_b64:
            raise ValueError('Capsule has no public_key field — cannot verify')
        verify_payload(canonical, sig, vk_b64)   # raises on tamper

    # ... rest of import logic unchanged ...

4. P3 — Session Continuity
This is the core value proposition of Compatible Companion. A unit written in session 1 must be present in session 2 without replay. The companion accumulates across separate AgentRuntime instantiations on the same DB. This requires no new code — it is a proof that the existing architecture works correctly. Write the test.

4.1 Acceptance Test (Test 28)
def test_28_session_continuity():
    """
    Session 2 sees session 1 writes without replay.
    This is the core CC value proposition:
    interaction state persists across runtime instantiations.
    """
    import tempfile, os
    from cc.runtime import AgentRuntime

    with tempfile.TemporaryDirectory() as tmp:
        db  = os.path.join(tmp, 'companion.db')
        log = os.path.join(tmp, 'receipts.jsonl')

        # ── Session 1 ──────────────────────────────────────────────
        rt1 = AgentRuntime(db_path=db, log_path=log)
        rt1.run_cycle(
            plan={'action':'mem_store','unit_id':'pref_voice',
                  'scope':'style','tags':['voice'],
                  'body':{'summary':'terse, direct, no hedging'}},
            act_fn=lambda p: {**p}, invariants=[],
        )
        rt1.run_cycle(
            plan={'action':'mem_store','unit_id':'pref_format',
                  'scope':'style','tags':['format'],
                  'body':{'summary':'bullet points for lists, prose otherwise'}},
            act_fn=lambda p: {**p}, invariants=[],
        )
        del rt1   # session 1 ends — runtime discarded

        # ── Session 2 — fresh runtime, same DB ──────────────────────
        rt2 = AgentRuntime(db_path=db, log_path=log)
        units = rt2.memory.mem_find(scope='style')
        ids = [u.get('unit_id') for u in units]

        assert 'pref_voice' in ids,  'session 1 voice pref must survive to session 2'
        assert 'pref_format' in ids, 'session 1 format pref must survive to session 2'

        # ── Session 3 — accumulate further ──────────────────────────
        rt3 = AgentRuntime(db_path=db, log_path=log)
        rt3.run_cycle(
            plan={'action':'mem_store','unit_id':'fact_001',
                  'scope':'core','tags':['bio'],
                  'body':{'summary':'user works in distributed systems'}},
            act_fn=lambda p: {**p}, invariants=[],
        )
        del rt3

        rt4 = AgentRuntime(db_path=db, log_path=log)
        all_units = rt4.memory.mem_find(scope='style') + rt4.memory.mem_find(scope='core')
        all_ids = [u.get('unit_id') for u in all_units]
        assert 'pref_voice' in all_ids,  'must accumulate across 3 sessions'
        assert 'pref_format' in all_ids, 'must accumulate across 3 sessions'
        assert 'fact_001' in all_ids,    'must accumulate across 3 sessions'
