2. W11-P1 — Data-Driven Rule Engine
The AST whitelist blocked known injection vectors. But eval is still eval — a Python string executing in a Python interpreter. The only complete fix is to never call eval. Replace string predicates with structured rule objects and a pure-Python evaluator.

2.1 New rule schema

2.3 Evaluator — add to cc/policy_store.py
# cc/policy_store.py — add below existing code

def evaluate_rule(rule: dict, unit: dict) -> bool:
    """
    Pure Python rule evaluator. No eval. No exec. No string execution.
    Takes a structured rule dict and a unit dict.
    Returns True if the rule passes (unit is allowed).
    """
    op = rule.get('operator')

    # Compound operators
    if op is None and 'all' in rule:
        return all(evaluate_rule(r, unit) for r in rule['all'])
    if op is None and 'any' in rule:
        return any(evaluate_rule(r, unit) for r in rule['any'])

    field = rule.get('field')
    value = rule.get('value')

    # Resolve field from unit (support dot notation for body fields)
    def resolve(u, f):
        parts = f.split('.')
        cur = u
        for part in parts:
            if isinstance(cur, dict):
                cur = cur.get(part)
            else:
                return None
        return cur

    fval = resolve(unit, field) if field else None

    if op == 'eq':          return fval == value
    if op == 'neq':         return fval != value
    if op == 'exists':      return fval is not None
    if op == 'contains':    return value in fval if fval is not None else False
    if op == 'not_contains':return value not in fval if fval is not None else True
    if op == 'min_length':  return len(fval) >= value if fval is not None else False
    if op == 'max_length':  return len(fval) <= value if fval is not None else True
    if op == 'prefix':      return str(fval).startswith(value) if fval is not None else False

    raise ValueError(f'Unknown operator: {op}')

def _eval_policy_rules(self, staged_writes):
    if self.policy_mode == 'locked_down':
        return None

    rules = self.policy_store.get_active_rules()
    for unit in staged_writes:
        for rule_row in rules:
            import json
            try:
                rule = json.loads(rule_row['predicate'])
            except (json.JSONDecodeError, TypeError):
                # Legacy string predicate — reject it, do not eval
                return f"rule {rule_row['rule_id']}: legacy string predicate not supported"
            if not evaluate_rule(rule, unit):
                return f"rule {rule_row['rule_id']} blocked unit {unit.get('unit_id')}"
    return None

3. W11-P2 — TLS on TCP
The TCP server currently runs plaintext. Anyone on the network can read or replay traffic. TLS wraps the socket — clients that do not negotiate TLS are dropped before the auth handshake. For development, a self-signed cert. For production, the cert path is passed at server startup.

3.1 server.py — ssl_context parameter
# In cc/server.py — update start_tcp_server
import ssl

async def start_tcp_server(rt, key_manager,
                            host='127.0.0.1', port=7700,
                            token: str = None,
                            ssl_context=None):
    if token is None:
        raise ValueError('token is required.')
    server = await asyncio.start_server(
        lambda r, w: _handle_client(
            rt, None, r, w, token,
            rt.db_path, rt.log_path, key_manager
        ),
        host, port,
        ssl=ssl_context   # None = plaintext (dev), SSLContext = TLS (prod)
    )
    if hasattr(server, '_start_event'):
        server._start_event.set()
    async with server:
        await server.serve_forever()

# Helper — generate a self-signed cert for testing
def make_dev_ssl_context(certfile: str, keyfile: str) -> ssl.SSLContext:
    """
    Creates a server-side SSLContext from a self-signed cert.
    Generate with: openssl req -x509 -newkey rsa:2048 -keyout key.pem
                           -out cert.pem -days 365 -nodes -subj '/CN=localhost'
    """
    ctx = ssl.SSLContext(ssl.PROTOCOL_TLS_SERVER)
    ctx.load_cert_chain(certfile, keyfile)
    return ctx

4. W11-P3 — Per-Client Ed25519 Identity
The shared secret gives binary access — you either have the token or you don't. Per-client identity gives each client a named, auditable identity. The server maintains a trusted public key allowlist. The handshake is a challenge-response: server sends a random nonce, client signs it with its private key, server verifies with the stored public key.

4.1 Protocol change — challenge-response handshake
# New handshake replacing shared-secret auth
#
# Step 1 — Client sends its client_id (name) and public key:
# {"hello": "my-agent", "vk": "<base64-ed25519-public-key>"}
#
# Step 2 — Server checks vk against allowlist.
#          If trusted, sends a random challenge:
# {"challenge": "<32-byte-hex-nonce>"}
#
# Step 3 — Client signs the challenge with its private key:
# {"signature": "<base64-ed25519-signature>"}
#
# Step 4 — Server verifies signature with stored vk.
#          On success:
# {"ok": true, "session_id": "<uuid4>", "client_id": "my-agent"}
#          On failure:
# {"ok": false, "error": "unauthorized"}  then close.

async def _handle_client(rt, reader, writer, token, db_path,
                          log_path, key_manager, client_keys):
    line = await reader.readline()
    try:
        req = json.loads(line.decode())
    except Exception:
        writer.close(); await writer.wait_closed(); return

    # Legacy shared-secret path (backward compat)
    if 'auth' in req and client_keys is None:
        if req['auth'] != token:
            writer.write((json.dumps({'ok':False,'error':'unauthorized'})+'\n').encode())
            await writer.drain()
            writer.close(); await writer.wait_closed(); return
        # proceed to session

    # Per-client identity path
    elif 'hello' in req and client_keys is not None:
        client_id = req.get('hello')
        vk_b64    = req.get('vk')
        if client_id not in client_keys or client_keys[client_id] != vk_b64:
            writer.write((json.dumps({'ok':False,'error':'unauthorized'})+'\n').encode())
            await writer.drain()
            writer.close(); await writer.wait_closed(); return
        # Issue challenge
        nonce = secrets.token_hex(32)
        writer.write((json.dumps({'challenge': nonce})+'\n').encode())
        await writer.drain()
        # Await signature
        sig_line = await reader.readline()
        sig_req  = json.loads(sig_line.decode())
        sig_b64  = sig_req.get('signature')
        try:
            verify_payload(nonce.encode(), sig_b64, vk_b64)
        except Exception:
            writer.write((json.dumps({'ok':False,'error':'unauthorized'})+'\n').encode())
            await writer.drain()
            writer.close(); await writer.wait_closed(); return
    else:
        writer.close(); await writer.wait_closed(); return

    session_id = str(uuid.uuid4())
    writer.write((json.dumps({'ok':True,'session_id':session_id})+'\n').encode())
    await writer.drain()
    # ... rest of session loop unchanged ...

4.3 Acceptance Test (Test 42)
def test_42_per_client_identity():
    """Known client authenticates via challenge-response.
    Unknown client (not in allowlist) is rejected.
    """
    import asyncio, json, tempfile, os
    from cc.runtime import AgentRuntime
    from cc.identity import generate_keypair, KeyManager
    from cc.server import start_tcp_server
    from cc.policy_store import seal_policy
    import shutil, base64
    import nacl.signing

    async def run():
        with tempfile.TemporaryDirectory() as tmp:
            # Server identity
            generate_keypair()
            km = KeyManager(shutil.copy2('cc_identity.key',
                            os.path.join(tmp,'id.key')))
            rt = AgentRuntime(db_path=os.path.join(tmp,'s.db'),
                              log_path=os.path.join(tmp,'r.jsonl'))
            seal_policy(rt.policy_store, rt.log_path)
            rt._recheck_policy_integrity()

            # Known client keypair
            sk = nacl.signing.SigningKey.generate()
            vk_b64 = base64.b64encode(bytes(sk.verify_key)).decode()
            client_keys = {'trusted-agent': vk_b64}

            started = asyncio.Event()
            task = asyncio.create_task(
                start_tcp_server(rt, km, host='127.0.0.1', port=7730,
                                 client_keys=client_keys, start_event=started))
            await started.wait()

            async def do_handshake(client_sk, client_id, client_vk):
                r, w = await asyncio.open_connection('127.0.0.1', 7730)
                w.write((json.dumps({'hello':client_id,'vk':client_vk})+'\n').encode())
                await w.drain()
                challenge_resp = json.loads((await r.readline()).decode())
                if 'challenge' not in challenge_resp:
                    w.close(); await w.wait_closed()
                    return challenge_resp
                nonce = challenge_resp['challenge'].encode()
                signed = client_sk.sign(nonce)
                sig_b64 = base64.b64encode(signed.signature).decode()
                w.write((json.dumps({'signature':sig_b64})+'\n').encode())
                await w.drain()
                result = json.loads((await r.readline()).decode())
                w.close(); await w.wait_closed()
                return result

            # Known client — must succeed
            resp = await do_handshake(sk, 'trusted-agent', vk_b64)
            assert resp['ok'], f'known client must auth: {resp}'

            # Unknown client — must be rejected
            evil_sk = nacl.signing.SigningKey.generate()
            evil_vk = base64.b64encode(bytes(evil_sk.verify_key)).decode()
            resp2 = await do_handshake(evil_sk, 'evil-agent', evil_vk)
            assert not resp2.get('ok'), f'unknown client must be rejected: {resp2}'

            task.cancel()
            try: await task
            except asyncio.CancelledError: pass

    asyncio.run(run())

# cc/daemon.py
import os, sys, signal, asyncio
from pathlib import Path

PID_FILE = Path.home() / '.compatible' / 'daemon.pid'

def daemon_start():
    if PID_FILE.exists():
        pid = int(PID_FILE.read_text().strip())
        try:
            os.kill(pid, 0)  # check alive
            print(f'Daemon already running (pid {pid})')
            return
        except ProcessLookupError:
            PID_FILE.unlink()  # stale PID

    pid = os.fork()
    if pid > 0:
        # Parent — write PID and exit
        PID_FILE.write_text(str(pid))
        print(f'Daemon started (pid {pid})')
        return

    # Child — become daemon
    os.setsid()
    sys.stdin  = open(os.devnull)
    sys.stdout = open(Path.home()/'.compatible'/'daemon.log', 'a')
    sys.stderr = sys.stdout

    from cc.cli import get_runtime, get_key_manager, DEFAULT_LOG
    from cc.server import start_tcp_server
    import json
    token = os.environ.get('CC_TOKEN', 'changeme')
    rt = get_runtime()
    km = get_key_manager()
    asyncio.run(start_tcp_server(rt, km, token=token))

def daemon_stop():
    if not PID_FILE.exists():
        print('No daemon running.')
        return
    pid = int(PID_FILE.read_text().strip())
    os.kill(pid, signal.SIGTERM)
    PID_FILE.unlink()
    print(f'Daemon stopped (pid {pid})')
